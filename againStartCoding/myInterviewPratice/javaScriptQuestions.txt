28. To convert all keys in an object (including deeply nested ones) to snake_case, 
here’s a JavaScript function that handles the transformation recursively:
function toSnakeCase(obj) { -> Paytm
  if (Array.isArray(obj)) {
    // If the value is an array, recursively call toSnakeCase on each element
    return obj.map(toSnakeCase);
  } else if (obj !== null && typeof obj === "object") {
    // If the value is an object, transform its keys
    return Object.keys(obj).reduce((acc, key) => {
      const snakeKey = key
        .replace(/([A-Z])/g, "_$1") // Insert underscore before uppercase letters
        .toLowerCase(); // Convert to lowercase
      acc[snakeKey] = toSnakeCase(obj[key]); // Recursively transform the value
      return acc;
    }, {});
  }
  // If not an object or array, return the value as is
  return obj;
}

function curry(func) { -> Paytm
  return function curried(...args) {
    // If enough arguments are provided, call the original function
    if (args.length >= func.length) {
      return func(...args);
    }
    // Otherwise, return a function that collects more arguments
    return function (...nextArgs) {
      return curried(...args, ...nextArgs); // Combine previous and new arguments
    };
  };
}

// Example usage
function sum(a, b, c) {
  return a + b + c;
}

let curriedSum = curry(sum);

// Test cases
console.log(curriedSum(1, 2, 3)); // Output: 6
console.log(curriedSum(1)(2, 3)); // Output: 6
console.log(curriedSum(1)(2)(3)); // Output: 6

41.// Function to add two numbers
function add(x, y) {
    return x + y;
}

// Function to format a number with a "$" sign
function format(x) {
    return "$" + x;
}

// Function to get formatted sum
function getProfit(a, b) {
    const sum = add(a, b);
    return format(sum); // Correctly formats the sum
}

// Function to compose two functions
function compose(f, g) {
    return function (...args) {
        return f(g(...args));
    };
}

// Composing `format` and `add`
const getFormattedSum = compose(format, add);

// Example usage
console.log(getFormattedSum(2, 5)); // Output: "$7"
